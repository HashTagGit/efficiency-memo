# 6.2 Poll a specific run until it completes
RUN_ID=123456789                 # from 6.1
for i in $(seq 1 30); do
  STATUS_JSON=$(curl -s \
    -H "Authorization: Bearer $GITHUB_TOKEN" \
    -H "Accept: application/vnd.github+json" \
    "https://api.github.com/repos/HashTagGit/efficiency-memo/actions/runs/$RUN_ID")
  STATE=$(echo "$STATUS_JSON" | jq -r '.status')
  CONCL=$(echo "$STATUS_JSON" | jq -r '.conclusion')
  echo "[$i] status=$STATE conclusion=$CONCL"
  if [ "$STATE" = "completed" ]; then break; fi
  sleep 30
done
